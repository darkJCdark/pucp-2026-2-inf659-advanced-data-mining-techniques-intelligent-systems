def top(idx, n=1):
    return list(idx[np.argsort(-conf[idx])][:n])

casos = [("Acierto", i) for i in top(ok[y_test[ok] == 1]) + top(ok[y_test[ok] == 0])] \

    for col, z in enumerate([16, 32, zmax]):
